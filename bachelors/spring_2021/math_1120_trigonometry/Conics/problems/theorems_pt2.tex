\begin{theorem}
    Suppose $e$ and $d$ are positive numbers. Then
    
    \begin{itemize}
        \item the graph $r = \frac{ed}{1 - e \cos(\theta)}$ is the graph of a conic section with directix $x = -d$ 
        \item the graph $r = \frac{ed}{1 + e \cos(\theta)}$ is the graph of a conic section with directix $x = d$
        \item the graph $r = \frac{ed}{1 - e \sin(\theta)}$ is the graph of a conic section with directix $y = -d$
        \item the graph $r = \frac{ed}{1 + e \sin(\theta)}$ is the graph of a conic section with directix $y = d$
    \end{itemize}
    
    In each case above, $(0,0)$ is a focus of the conic and the number $e$ is the eccentricity of the conic.
    
    \begin{itemize}
        \item If $0 < e < 1$, the graph is an ellipse whose major axis has length $\frac{2ed}{1 - e^2}$ and whose minor axis has length $\frac{2ed}{\sqrt{1 - e^2}}$
        \item If $e = 1$, the graph is a parabola whose focal diameter is $2d$. 
        \item If $e > 1$, the graph is an ellipse whose major axis has length $\frac{2ed}{e^2 - 1}$ and whose minor axis has length $\frac{2ed}{\sqrt{e^2 - 1}}$
    \end{itemize}
    
    \medskip \hrule \medskip
\end{theorem}
