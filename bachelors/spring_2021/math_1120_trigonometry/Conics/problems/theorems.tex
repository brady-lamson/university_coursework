\begin{theorem}[\textbf{Rotation of Axes}]
    Suppose the positive $x$ and $y$ axes are rotated counter-clockwise through an angle $\theta$ to produce the axes $x'$ and $y'$ respectively. Then the coordinates $P(x, y)$ and $P(x', y')$ are related by the following systems of equations. 
    
    \begin{equation*}
        \begin{cases}
            x = x'\cos(\theta) - y'\sin(\theta) \\
            y = x'\sin(\theta) + y'\cos(\theta)
        \end{cases}
    \end{equation*}
    
    \begin{equation*}
        \begin{cases}
            x' = x\cos(\theta) + y\sin(\theta) \\
            y' = -x\sin(\theta) + y\cos(\theta)
        \end{cases}
    \end{equation*}
\end{theorem}

\begin{theorem}
    The equation $Ax^2 + Bxy + Cy^2 + Dx + Ey + F = 0$ with $B \neq 0$ can be transformed into an equation with variables $x'$ and $y'$ without any $x' y'$ terms by rotating the x and y axes counter-clockwise through an angle $\theta$ which satisfies:
    \[cot(2\theta) = \frac{A - C}{B}\]
\end{theorem}

\begin{theorem}
    Suppose the equation $Ax^2 + Bxy + Cy^2 + Dx + Ey + F = 0$ describes a non-degenerate conic section. 
    
    \begin{itemize}
        \item If $B^2 - 4AC > 0$ then the graph of the equation is a hyperbola.
        \item If $B^2 - 4AC = 0$ then the graph of the equation is a parabola.
        \item If $B^2 - 4AC < 0$ then the graph of the equation is an ellipse or circle.
    \end{itemize}
    
    Note: A non-degenerate conic section refers to a graph that is either a circle, parabola, ellipse, or hyperbola. 
\end{theorem}
