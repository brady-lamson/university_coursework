\begin{problem}[26.]
    $( -117, -117\pi)$    
\end{problem}

\begin{solution}
    Note: $-117\pi$ radians is equivalent to $-\pi$ radians when calculating Cartesian coordinates! \medskip

    Calculating $x$:
    \begin{align}
        x &= r \cdot \cos(\theta) \\
        x &= -117 \cdot \cos \left( -117\pi \right) \\
        x &= -117 \cdot -1 \\
        x &= 117
    \end{align}
    
    Calculating $y$:
    \begin{align}
        y &= r \cdot \sin(\theta) \\
        y &= -117 \cdot \sin \left( -\pi \right) \\
        y &= -117 \cdot 0 \\
        y &= 0
    \end{align}
    
    \[\boxed{ (117, 0) }\]
\end{solution}
