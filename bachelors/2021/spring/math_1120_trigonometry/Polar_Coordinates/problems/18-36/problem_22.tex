\begin{problem}[22.]
    $\left( -4, \frac{5\pi}{6} \right)$    
\end{problem}

\begin{solution}
    Calculating $x$:
    \begin{align}
        i) \; x &= r \cdot \cos(\theta) && iii) \; x = -4 \cdot - \frac{\sqrt{3}}{2} \\
        ii) \; x &= -4 \cdot \cos \left( \frac{5\pi}{6} \right) && iv) \; \boxed{x = 2\sqrt{3}}
    \end{align}
    
    Calculating $y$:
    \begin{align}
        i) \; y &= r \cdot \sin(\theta) && iii) \; y = -4 \cdot \frac{1}{2} \\
        ii) \; y &= -4 \cdot \sin \left( \frac{5\pi}{6} \right) && iv) \; \boxed{y = -2} \\
    \end{align}
    
    \[\boxed{ (2\sqrt{3}, -2) }\]
\end{solution}
