\begin{problem}[18.]
    $\left( 2, \frac{\pi}{3} \right)$    
\end{problem}

\begin{solution}
    Calculating $x$:
    \begin{align}
        x &= r \cdot \cos(\theta) \\
        x &= 2 \cdot \cos \left( \frac{\pi}{3} \right) \\
        x &= 1
    \end{align}
    
    Calculating $y$:
    \begin{align}
        y &= r \cdot \sin(\theta) \\
        y &= 2 \cdot \sin \left( \frac{\pi}{3} \right) \\
        y &= 2 \cdot \frac{\sqrt{3}}{2} \\
        y &= \sqrt{3}
    \end{align}
    
    \[\boxed{ (1, \sqrt{3}) }\]
\end{solution}
