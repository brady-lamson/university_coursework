\begin{problem}[81.]
    $\theta = \frac{2\pi}{3}$
\end{problem}

\begin{solution}
    \begin{align}
        \theta &= \frac{2\pi}{3} \\
        \tan(\theta) &= \tan \left( \frac{2\pi}{3} \right) \\
        \tan(\theta) &= -\sqrt{3} \\
        \tan(\theta) &= \frac{y}{x} \\
        -\sqrt{3} &= \frac{y}{x}
    \end{align}
    \[\boxed{y = x \cdot -\sqrt{3} }\]
    \[\boxed{x = -\frac{y}{\sqrt{3}}}\]
\end{solution}
