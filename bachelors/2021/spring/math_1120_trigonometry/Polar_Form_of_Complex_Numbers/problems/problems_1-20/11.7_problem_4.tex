\begin{problem}[4.]
    $ z = -3\sqrt{2} + 3 i \sqrt{2}$
\end{problem}

\begin{solution}
    Please refer to problem 1 for more detailed logic. 
    
    \[z = a + bi\]
    \[a = -3\sqrt{2} \; \text{and} \; b = 3\sqrt{2} \]
    \[ \boxed{ -3\sqrt{2} = Re(z)} \; \text{and} \; \boxed{ 3\sqrt{2} = Im(z) }\]
    
    Calculating $r$ and $|z|$:
    \begin{align}
        a^2 + b^2 = r^2 \\
        (-3\sqrt{2})^2 + (3\sqrt{2})^2 = r^2 \\
        \sqrt{36} = r \\
        \boxed{\pm 6 = r} \\ 
        \boxed{6 = |z|}
    \end{align}
    
    Calculating $\theta$:
    \begin{align}
        \tan(\theta) &= \frac{Im(z)}{Re(z)} = \frac{b}{a} \\
        \tan(\theta) &= -\frac{ 3\sqrt{2} }{ 3\sqrt{2} } \\ 
        \tan(\theta) &= -1 \\
        \theta &= \frac{3\pi}{4} + \pi k \; \text{for integers k}
    \end{align}
    \[ \boxed{arg(z) = \left\{ \frac{3\pi}{4} + \pi k | \; \text{k is an integer} \right\} } \]
    
    $\frac{3\pi}{4}$ is between $-\pi$ and $\pi$ so we can say:
    \[ \boxed{Arg(z) = \frac{3\pi}{4} }\]
    
    \[\boxed{ \text{Polar Rep.} \; = 6 \; cis \left( \frac{3\pi}{4} \right) }\]
\end{solution}
